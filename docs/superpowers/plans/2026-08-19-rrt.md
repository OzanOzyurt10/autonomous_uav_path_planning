```python
def assert_pose_close(actual, expected, tol=1e-6):
    assert math.isclose(actual[0], expected[0], abs_tol=tol), f"x: {actual} vs {expected}"
    assert math.isclose(actual[1], expected[1], abs_tol=tol), f"y: {actual} vs {expected}"
    dyaw = (actual[2] - expected[2]) % (2 * math.pi)
    dyaw = min(dyaw, 2 * math.pi - dyaw)
    assert dyaw < tol, f"yaw: {actual} vs {expected}"


class TestNode:
    def test_fields_are_positional_in_order(self):
        path = shortest_path(START, GOAL, RHO)
        node = Node((1.0, 2.0, 0.5), 3, 12.5, path)
        assert node.pose == (1.0, 2.0, 0.5)
        assert node.parent == 3
        assert node.cost == 12.5
        assert node.path_from_parent is path

    def test_root_has_no_parent(self):
        root = Node(START, None, 0.0, None)
        assert root.parent is None
        assert root.path_from_parent is None
        assert root.cost == 0.0

    def test_is_frozen(self):
        node = Node(START, None, 0.0, None)
        with pytest.raises(Exception):
            node.cost = 5.0


class TestRRTResult:
    def test_fields_are_positional_in_order(self):
        edges = [shortest_path(START, GOAL, RHO)]
        tree = [Node(START, None, 0.0, None)]
        result = RRTResult(True, edges, 90.0, 17, tree)
        assert result.found is True
        assert result.edges == edges
        assert result.cost == 90.0
        assert result.iterations == 17
        assert result.tree == tree

    def test_failure_shape(self):

        result = RRTResult(False, [], math.inf, 5000, tree)
        assert result.found is False
        assert result.edges == []
        assert result.cost == math.inf
        assert result.tree == tree
```

```python
from src.rrt_star import _nearest


def _node(pose):
    return Node(pose, None, 0.0, None)


class TestNearest:
    def test_single_node_returns_zero(self):
        assert _nearest([_node(START)], GOAL, RHO) == 0

    def test_picks_minimum_dubins_distance(self):
        nodes = [_node((0.0, 0.0, 0.0)),
                 _node((50.0, 0.0, 0.0)),
                 _node((90.0, 0.0, 0.0))]
        target = (95.0, 0.0, 0.0)
        assert _nearest(nodes, target, RHO) == 2

    def test_dubins_not_euclidean(self):
        """Oklid'e gore yakin ama ters bakan dugum secilmemeli."""
        target = (10.0, 0.0, 0.0)
        near_but_backwards = _node((5.0, 0.0, math.pi))   # oklid 5, dubins ~13
        far_but_aligned = _node((0.0, 0.0, 0.0))          # oklid 10, dubins 10
        nodes = [near_but_backwards, far_but_aligned]
        assert _nearest(nodes, target, 2.0) == 1

    def test_direction_is_from_tree_to_target(self):
        """Mesafe gidis yonunde olculmeli; Dubins simetrik degildir."""
        # P=(0,0,0) -> Q=(0,3,pi/2) = 14.857 m,  Q -> P = 11.661 m
        p, q = (0.0, 0.0, 0.0), (0.0, 3.0, math.pi / 2)
        assert not math.isclose(path_length(p, q, 2.0), path_length(q, p, 2.0))

        # agacta P ve uzak bir dugum var; hedef Q.
        # dogru yon P->Q olculmeli
        far = _node((60.0, 60.0, 0.0))
        nodes = [_node(p), far]
        assert _nearest(nodes, q, 2.0) == 0

    def test_returns_index_not_node(self):
        result = _nearest([_node(START), _node(GOAL)], GOAL, RHO)
        assert isinstance(result, int)

    def test_ties_pick_first(self):
        nodes = [_node((0.0, 0.0, 0.0)), _node((0.0, 0.0, 0.0))]
        assert _nearest(nodes, (10.0, 0.0, 0.0), RHO) == 0
```

- [ ] **Adım 3: `_nearest`'i yaz — OZAN**

Tam imza:
```python
def _nearest(nodes: list[Node], target: Pose, rho: float) -> int:
```

**Sözleşme:** `path_length(nodes[i].pose, target, rho)` değeri en küçük olan
`i` indeksini döndürür. **Düğümü değil, indeksi.** Eşitlikte ilk gelen kazanır.

**Yön kritik:** `path_length(dugum, target, rho)` — ağaçtan hedefe. Ters yazarsan
(`path_length(target, dugum, rho)`) test hemen yakalamayabilir ama ağaç yanlış
büyür. Dubins mesafesi simetrik değildir: `(0,0,0) → (0,3,π/2)` 14.86 m,
tersi 11.66 m.

**Öklid mesafesi kullanma.** Yakın ama ters yöne bakan bir düğüme bağlanmak
uzun bir dönüş gerektirir; Dubins bunu bilir, `hypot` bilmez.
`test_dubins_not_euclidean` tam bunu sınıyor.

`min()` bir `key` fonksiyonuyla en küçüğü seçer, `range(len(nodes))` üzerinde
çalıştırırsan indeks alırsın:
```python
min(range(len(nodes)), key=lambda i: ...)
```
`min` eşitlikte ilk bulduğu elemanı döndürür — `test_ties_pick_first` bunun
üstüne kuruluyor. Düz `for` döngüsüyle yazarsan da olur, o zaman
karşılaştırmayı `<` ile yap (`<=` yazarsan eşitlikte sonuncuyu seçersin).

Tam imza:
```python
def _sample(env: Environment, goal: Pose, rng: random.Random,
            goal_bias: float) -> Pose:
```

**Sözleşme:** `goal_bias` olasılıkla doğrudan `goal` pozunu, geri kalan
durumlarda `env.random_free_pose(rng)` döndürür.

`rng.random()` `[0.0, 1.0)` aralığında ondalık sayı verir. `goal_bias`'tan
küçükse hedefi döndür.

**Karşılaştırma yönüne dikkat:** `rng.random() < goal_bias` yazmalısın.
`goal_bias = 0.0` olduğunda hiçbir sayı `0.0`'dan küçük olamaz, yani hedef
hiç dönmez — `test_zero_bias_never_returns_goal` bunu istiyor. `<=` yazarsan
`rng.random()` bir gün tam `0.0` üretir ve test kırılır.
`goal_bias = 1.0` olduğunda ise `[0,1)` aralığındaki her sayı `1.0`'dan küçük,
yani hep hedef döner.

**Neden hedef biasing var:** Ağacı hedefe doğru çekiyor. Olmadan RRT serbest
alanda amaçsızca yayılır; hedefe rastgele denk gelmeyi bekler ve yakınsama
çok yavaşlar. 0.05 yaygın bir başlangıç değeri: iterasyonların %5'i doğrudan
hedefe uzanmayı dener, %95'i keşfe ayrılır.

```python
from src.rrt_star import _extract_path


class TestExtractPath:
    def _chain(self):
        """Uc dugumluk zincir: A -> B -> C, ve C'den hedefe bir kenar."""
        a = (0.0, 0.0, 0.0)
        b = (20.0, 0.0, 0.0)
        c = (40.0, 0.0, 0.0)
        ab = shortest_path(a, b, RHO)
        bc = shortest_path(b, c, RHO)
        goal_edge = shortest_path(c, GOAL, RHO)
        nodes = [
            Node(a, None, 0.0, None),
            Node(b, 0, ab.length, ab),
            Node(c, 1, ab.length + bc.length, bc),
        ]
        return nodes, goal_edge, [ab, bc, goal_edge]

    def test_returns_edges_in_travel_order(self):
        nodes, goal_edge, expected = self._chain()
        assert _extract_path(nodes, 2, goal_edge) == expected

    def test_first_edge_starts_at_root(self):
        nodes, goal_edge, _ = self._chain()
        edges = _extract_path(nodes, 2, goal_edge)
        assert edges[0].start == nodes[0].pose

    def test_last_edge_is_goal_edge(self):
        nodes, goal_edge, _ = self._chain()
        assert _extract_path(nodes, 2, goal_edge)[-1] is goal_edge

    def test_chain_is_continuous(self):
        nodes, goal_edge, _ = self._chain()
        edges = _extract_path(nodes, 2, goal_edge)
        for first, second in zip(edges, edges[1:]):
            assert_pose_close(first.end_pose(), second.start)

    def test_from_root_gives_only_goal_edge(self):
        nodes, goal_edge, _ = self._chain()
```

- [ ] **Adım 3: `_extract_path`'i yaz — OZAN**

Tam imza:
```python
def _extract_path(nodes: list[Node], index: int,
                  goal_edge: DubinsPath) -> list[DubinsPath]:
```

**Sözleşme:** `nodes[index]`'ten köke kadar ebeveyn zincirini geri takip eder,
toplanan kenarları **gidiş sırasına** çevirir ve sonuna `goal_edge`'i ekler.

Algoritma:
1. Boş liste aç
2. `index`'ten başla; düğümün `parent`'ı `None` olana kadar:
   - `path_from_parent`'ı listeye ekle
   - `index = nodes[index].parent`
3. Liste şu an **kökten geriye** sıralı — ters çevir
4. Sonuna `goal_edge`'i ekle, döndür

**Neden ters çevirmek gerekiyor:** Zinciri yapraktan köke doğru yürüyorsun,
yani kenarları ters sırada topluyorsun. Uçak ise kökten yaprağa gidecek.

Listeyi ters çevirmenin iki yolu: `liste.reverse()` listeyi **yerinde** çevirir
ve `None` döndürür (`return liste.reverse()` yazarsan `None` dönersin — dikkat);
`liste[::-1]` ise ters çevrilmiş **yeni** bir liste verir.

Kök düğümde `path_from_parent` `None`'dır ve zaten döngüye girmez, çünkü
`parent is None` koşulu orada durur.

- [ ] **Adım 4: Ana döngüyü yaz — OZAN**

Doğrulamadan sonra:

```
1. kok dugumu olustur:  Node(start, None, 0.0, None)
   nodes = [kok]
   rng None ise random.Random()

2. iteration 1'den max_iterations'a kadar:
     ornek = _sample(env, goal, rng, goal_bias)
     i     = _nearest(nodes, ornek, rho)
     yol   = _try_connect(env, nodes[i].pose, ornek, rho, step)
     yol None ise -> devam et (continue)

     yeni = Node(ornek, i, nodes[i].cost + yol.length, yol)
     nodes'a ekle

     hedef_yolu = _try_connect(env, ornek, goal, rho, step)
     hedef_yolu varsa ve henuz cozum kaydedilmemisse:
         edges = _extract_path(nodes, len(nodes) - 1, hedef_yolu)
         cozumu kaydet: (edges, yeni.cost + hedef_yolu.length, iteration)
         stop_on_first_solution ise -> RRTResult(True, ...) dondur

3. dongu bitti:
   kaydedilmis cozum varsa -> RRTResult(True, kaydedilen..., max_iterations, nodes)
   yoksa                   -> RRTResult(False, [], math.inf, max_iterations, nodes)
```

**`stop_on_first_solution` nasıl işliyor:** `True` (varsayılan) ise ilk çözümde
hemen dönülür. `False` ise çözüm bir kenara yazılır ve döngü bütçe boyunca
sürer; düz RRT rotayı iyileştirmediği için sonunda **aynı ilk çözüm** döner,
tek farkı `tree`'nin daha büyük olması. Parametre şimdiden imzada duruyor ki
2. aşamada `*` eklenince API değişmesin — o zaman `False` gerçekten anlam
kazanacak, çünkü yeniden bağlama rotayı kısaltacak.

`iterations` alanı çözümün **bulunduğu** iterasyondur; `False` ile bütçe
sonuna kadar dönülse bile kaydedilen değer korunur mu, yoksa `max_iterations`
mı yazılır — bu aşamada `stop_on_first_solution=True` varsayılan olduğu için
testler yalnızca hemen dönen yolu sınıyor. `False` durumunda `max_iterations`
yaz.

**Dört ayrıntı:**

`_extract_path`'e verilen indeks **yeni eklenen düğümün** indeksi, yani
`len(nodes) - 1`. Ekleme yapmadan önce hesaplarsan bir eksik alırsın.

`iterations` alanı **kaçıncı iterasyonda bulunduğu**. `for iteration in
range(1, max_iterations + 1)` yazarsan doğrudan elinde olur.
`test_iterations_is_one` boş haritada 1 bekliyor.

`continue` bir sonraki iterasyona atlar. `yol is None` durumunda bu iterasyon
boşa gitmiştir — bağlanamadık, ağaç büyümedi.

Toplam maliyet `yeni.cost + hedef_yolu.length`. `yeni.cost` zaten kökten oraya
kadarki toplam; üstüne son kenarı ekliyorsun.

Docstring yaz: parametreler, dönüş değeri, fırlatılan istisnalar ve
`found=False`'ın istisna olmadığı.
